```
from datetime import datetime

s = "14:30:05"
# %H:%M:%S 告诉 Python：第一个数字是时，第二个是分，第三个是秒
res = datetime.strptime(s, "%H:%M:%S")

print(res)        # 输出: 1900-01-01 14:30:05
print(res.hour)   # 输出: 14
```


## 时间计算
```
import os

import sys

# 请在此输入您的代码

from datetime import datetime,timedelta

start=datetime(2000,1,1)

end=datetime(2020,10,1)

  

cnt=0

while start<=end:

  if start.weekday()==0 or start.day==1:  #星期一或者月初

    cnt+=2

  else:

    cnt+=1


  start+=timedelta(days=1)   #days,seconds

  
print(cnt)
```

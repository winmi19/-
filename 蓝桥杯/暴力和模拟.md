s=input()

n=len(s)

delete=[False]*(n)

  

while True:

  for i in range(1,len(s)-1):

    if s[i]==s[i+1] and s[i]!=s[i-1]:

      delete[i]=delete[i-1]=True

  

    if s[i]==s[i-1] and s[i]!=s[i+1]:

      delete[i]=delete[i+1]=True

  

  new_s=""

  for i in range(len(s)):

    if delete==False:

      new_s+=s[i]

  

  #判断退出：新的字符串和旧的一样

  if s==new_s:

    break

  s=new_s

  

if s:

  print(s)

else:

  print('EMPTY')

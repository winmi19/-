def Two_sum(nums:list[int],t:int):

 d={}

 res=[]

 for i,x in enumerate(nums):

    if d.get(t-x) is not None:

        res.append([i,d.get(t-x)])

    d[x]=i  #字典里面存储索引，返回索引对

  

 return res

  

nums = [3, 2, 4, 3]

t = 6

res=Two_sum(nums,t)

for x in res:

   x.sort()

res.sort()

print(res)   #对列表两层进行升序

    torch.rand(*size, *, out=None, dtype=None, layout=torch.strided,

                device=None, requires_grad=False) -> Tensor

功能：

    生成指定形状的张量，元素值服从均匀分布 U[0, 1)

参数说明：

    *size：张量的形状，可以是多个整数，如 (3, 4) 或 3, 4

    dtype：数据类型，默认为 torch.float32

    device：设备，'cpu' 或 'cuda'

    requires_grad：是否需要计算梯度（用于自动求导）

数学原理：

    概率密度函数：f(x) = 1, 当 0 ≤ x < 1

    期望值：E[X] = 0.5

    方差：Var[X] = 1/12 ≈ 0.0833

应用场景：

    1. 初始化权重（需要后续缩放）

    3. 随机选择索引

def generate_synthetic_dataset(n_samples=100, n_features=5, n_classes=3, seed=42):

    """

    生成合成数据集

    参数：

        n_samples: 样本数量

        n_features: 特征数量

        n_classes: 类别数量

        seed: 随机种子

    返回：

        X: 特征矩阵 (n_samples, n_features)

        y: 标签向量 (n_samples,)

    """

    # 设置随机种子确保可重复性

    torch.manual_seed(seed)

    # 生成特征矩阵：每个特征来自不同的正态分布

    X = torch.randn(n_samples, n_features)

    # 为每个类别生成不同的中心点

    class_centers = torch.randn(n_classes, n_features) * 2.0

    # 生成标签：基于最近的中心点

    y = torch.randint(0, n_classes, (n_samples,))

    # 根据标签调整特征，使不同类别有区别

    for i in range(n_samples):

        X[i] += class_centers[y[i]] * 0.5

    return X, y

def random_data_augmentation(images, p_dropout=0.1, noise_std=0.05):

    """

    对图像批次进行随机数据增强

    参数：

        images: 输入图像批次 (batch, channels, height, width)

        p_dropout: Dropout概率

        noise_std: 高斯噪声标准差

    """

    batch_size = images.shape[0]

    augmented = images.clone()

    # 1. 随机水平翻转（每张图片独立决定）

    flip_mask = torch.rand(batch_size) < 0.5  # 50%概率翻转

    for i in range(batch_size):

        if flip_mask[i]:

            augmented[i] = torch.flip(augmented[i], dims=[-1])  # 水平翻转

    # 2. 随机添加高斯噪声

    noise = torch.randn_like(augmented) * noise_std

    augmented = augmented + noise

    # 3. 随机亮度调整

    brightness_factor = torch.empty(batch_size, 1, 1, 1).uniform_(0.8, 1.2)

    augmented = augmented * brightness_factor

    # 4. 随机Dropout（模拟遮挡）

    dropout_mask = torch.rand_like(augmented) > p_dropout

    augmented = augmented * dropout_mask.float()

    return augmented

    """

    生成混合分布的随机数
